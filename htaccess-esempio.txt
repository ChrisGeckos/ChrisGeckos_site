# ESEMPIO per hosting Apache: rinominare in ".htaccess" e metterlo nella cartella principale del sito.
# Porta tutto su https://www.chrisgeckos.it (indirizzo principale) con redirect permanente 301:
# dominio .com, versione senza www e versione http.
RewriteEngine On
RewriteCond %{HTTP_HOST} ^(www\.)?chrisgeckos\.com$ [NC,OR]
RewriteCond %{HTTP_HOST} ^chrisgeckos\.it$ [NC,OR]
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://www.chrisgeckos.it/$1 [R=301,L]

# Velocità: compressione e cache dei file (richiede i moduli mod_deflate e mod_expires, di solito attivi)
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType image/png "access plus 1 year"
  ExpiresByType image/jpeg "access plus 1 year"
  ExpiresByType text/css "access plus 1 week"
  ExpiresByType application/javascript "access plus 1 week"
</IfModule>
